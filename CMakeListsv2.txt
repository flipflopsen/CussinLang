# -----------------------------------------------------------------------------
# Targets
# -----------------------------------------------------------------------------

add_library(core_lib STATIC ${CORE_SOURCES})

target_include_directories(
    core_lib
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/src"
        ${LLVM_INCLUDE_DIRS}
)

target_link_libraries(
    core_lib
    PUBLIC
        ${llvm_libs}
)

add_executable(
    CussingLangImpl
    ${APP_SOURCES}
)

target_link_libraries(
    CussingLangImpl
    PRIVATE
        core_lib
)

add_executable(
    UnitTests
    ${TEST_SOURCES}
)

target_link_libraries(
    UnitTests
    PRIVATE
        core_lib
        GTest::gtest
        GTest::gtest_main
        GTest::gmock
        GTest::gmock_main
)

# /FR is an MSVC-specific source-browser-information option. Do not pass it to
# GCC, Clang, or other compilers.
if(MSVC)
    target_compile_options(core_lib PRIVATE /FR)
    target_compile_options(CussingLangImpl PRIVATE /FR)
    target_compile_options(UnitTests PRIVATE /FR)
endif()
